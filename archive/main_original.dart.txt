  try {
    await Firebase.initializeApp(
      options: DefaultFirebaseOptions.currentPlatform,

    final auth = FirebaseAuth.instance;

    );

// MODELOS: cada clase representa una entidad del dominio.
class Boletin {
  final String id, titulo, resumen, zona, categoria, autorId;
  final DateTime fecha;

  const Boletin({
    required this.id,
    required this.titulo,
    required this.resumen,
    required this.zona,
    required this.categoria,
    this.autorId = '',
    required this.fecha,
  });
  factory Boletin.fromDoc(DocumentSnapshot<Map<String, dynamic>> doc) {
    final d = doc.data()!;
    return Boletin(
      id: doc.id,
      titulo: d['titulo'] as String? ?? '',
      resumen: d['resumen'] as String? ?? '',
      zona: d['zona'] as String? ?? '',
      categoria: d['categoria'] as String? ?? '',
      autorId: d['autorId'] as String? ?? '',
      fecha:
          (d['fecha'] as Timestamp?)?.toDate() ??
          DateTime.fromMicrosecondsSinceEpoch(0),
    );
  }
}

// ESTADO COMPARTIDO: encapsula listas y operaciones del proyecto.
class PublinfoStore extends ChangeNotifier {
  PublinfoStore(this._firestore) {
    _boletinesSub = _firestore
        .collection('boletines')
        .snapshots()
        .listen(_cargarBoletines, onError: _registrarError);
    _comerciosSub = _firestore
        .collection('comercios')
        .snapshots()
        .listen(_cargarComercios, onError: _registrarError);
    _anunciosSub = _firestore
        .collection('anuncios')
        .snapshots()
        .listen(_cargarAnuncios, onError: _registrarError);
  }

  final FirebaseFirestore _firestore;
  final List<Boletin> _boletines = [];
  final List<Comercio> _comercios = [];
  final List<Anuncio> _anuncios = [];
  late final StreamSubscription<QuerySnapshot<Map<String, dynamic>>>
  _boletinesSub;
  late final StreamSubscription<QuerySnapshot<Map<String, dynamic>>>
  _comerciosSub;
  late final StreamSubscription<QuerySnapshot<Map<String, dynamic>>>
  _anunciosSub;

  List<Boletin> get boletines => List.unmodifiable(_boletines);
  List<Comercio> get comercios => List.unmodifiable(_comercios);
  List<Anuncio> get anuncios => List.unmodifiable(_anuncios);

  void _registrarError(Object error) {
    debugPrint('Error de Firestore: $error');
  }

  void _cargarBoletines(QuerySnapshot<Map<String, dynamic>> snapshot) {
    _boletines
      ..clear()
      ..addAll(snapshot.docs.map(Boletin.fromDoc));
    _boletines.sort((a, b) => b.fecha.compareTo(a.fecha));
    notifyListeners();
  }

  void _cargarComercios(QuerySnapshot<Map<String, dynamic>> snapshot) {
    _comercios
      ..clear()
      ..addAll(snapshot.docs.map(Comercio.fromDoc));
    _comercios.sort((a, b) => a.nombre.compareTo(b.nombre));
    notifyListeners();
  }

  void _cargarAnuncios(QuerySnapshot<Map<String, dynamic>> snapshot) {
    _anuncios
      ..clear()
      ..addAll(snapshot.docs.map(Anuncio.fromDoc));
    _anuncios.sort((a, b) => b.fecha.compareTo(a.fecha));
    notifyListeners();
  }

  Comercio? buscarComercio(String id) {
    for (final comercio in _comercios) {
      if (comercio.id == id) return comercio;
    }
    return null;
  }

  String get _uid {
    final uid = FirebaseAuth.instance.currentUser?.uid;
    if (uid == null || uid.isEmpty) {
      throw StateError('No hay una sesión autenticada.');
    }
    return uid;
  }

  bool esPropietario(String autorId) =>
      autorId.isNotEmpty && autorId == FirebaseAuth.instance.currentUser?.uid;

  Future<void> agregarBoletin({
    required String titulo,
    required String resumen,
    required String zona,
    required String categoria,
  }) async {
    await _firestore.collection('boletines').add({
      'titulo': titulo.trim(),
      'resumen': resumen.trim(),
      'zona': zona.trim(),
      'categoria': categoria.trim(),
      'autorId': _uid,
      'fecha': FieldValue.serverTimestamp(),
    });
  }

  Future<void> eliminarBoletin(String id) =>
      _firestore.collection('boletines').doc(id).delete();

  Future<void> agregarComercio({
    required String nombre,
    required String direccion,
    required String zona,
    required String categoria,
    required String telefono,
  }) async {
    await _firestore.collection('comercios').add({
      'nombre': nombre.trim(),
      'direccion': direccion.trim(),
      'zona': zona.trim(),
      'categoria': categoria.trim(),
      'telefono': telefono.trim(),
      'autorId': _uid,
    });
  }

  Future<void> eliminarComercio(String id) async {
    final anuncios = await _firestore
        .collection('anuncios')
        .where('comercioId', isEqualTo: id)
        .get();
    if (anuncios.docs.isNotEmpty) {
      final batch = _firestore.batch();
      for (final doc in anuncios.docs) {
        batch.delete(doc.reference);
      }
      await batch.commit();
    }
    await _firestore.collection('comercios').doc(id).delete();
  }

  Future<void> agregarAnuncio({
    required String comercioId,
    required String titulo,
    required String descripcion,
  }) async {
    if (buscarComercio(comercioId) == null) {
      throw ArgumentError('El comercio seleccionado no existe.');
    }
    await _firestore.collection('anuncios').add({
      'comercioId': comercioId,
      'titulo': titulo.trim(),
      'descripcion': descripcion.trim(),
      'fecha': FieldValue.serverTimestamp(),
      'autorId': _uid,
    });
  }

  Future<void> eliminarAnuncio(String id) =>
      _firestore.collection('anuncios').doc(id).delete();

  @override
  void dispose() {
    _boletinesSub.cancel();
    _comerciosSub.cancel();
    _anunciosSub.cancel();
    super.dispose();
  }
}

String fechaCorta(DateTime fecha) {
  final local = fecha.toLocal();
  String dos(int n) => n.toString().padLeft(2, '0');
  return '${dos(local.day)}/${dos(local.month)}/${local.year} ${dos(local.hour)}:${dos(local.minute)}';
}

void mostrarDetalle(BuildContext context, String titulo, List<String> lineas) {
  showDialog<void>(
    context: context,
    builder: (dialogContext) => AlertDialog(
      title: Text(titulo),
      content: SingleChildScrollView(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            for (final linea in lineas)
              Padding(
                padding: const EdgeInsets.only(bottom: 10),
                child: Text(linea),
              ),
          ],
        ),
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.pop(dialogContext),
          child: const Text('Cerrar'),
        ),
      ],
    ),
  );
}

class BoletinesTab extends StatelessWidget {
  final PublinfoStore store;
  const BoletinesTab({super.key, required this.store});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Boletines locales')),
      body: ListenableBuilder(
        listenable: store,
        builder: (context, _) {
          final boletines = store.boletines;
          if (boletines.isEmpty) {
            return const Center(child: Text('Todavía no hay boletines.'));
          }
          return ListView.builder(
            padding: const EdgeInsets.all(12),
            itemCount: boletines.length,
            itemBuilder: (context, index) {
              final boletin = boletines[index];
              return Card(
                child: ListTile(
                  isThreeLine: true,
                  title: Text(boletin.titulo),
                  subtitle: Text(
                    '${boletin.categoria} · ${boletin.zona}\n${boletin.resumen}',
                    maxLines: 3,
                    overflow: TextOverflow.ellipsis,
                  ),
                  onTap: () => mostrarDetalle(context, boletin.titulo, [
                    'Categoría: ${boletin.categoria}',
                    'Zona: ${boletin.zona}',
                    'Fecha: ${fechaCorta(boletin.fecha)}',
                    boletin.resumen,
                  ]),
                  trailing: store.esPropietario(boletin.autorId)
                      ? IconButton(
                          tooltip: 'Eliminar boletín',
                          icon: const Icon(Icons.delete_outline),
                          onPressed: () => confirmarEliminacion(
                            context,
                            'boletín',
                            () => store.eliminarBoletin(boletin.id),
                          ),
                        )
                      : null,
                ),
              );
            },
          );
        },
      ),
      floatingActionButton: FloatingActionButton.extended(
        onPressed: () => Navigator.push(
          context,
          MaterialPageRoute<void>(
            builder: (_) => FormularioBoletin(store: store),
          ),
        ),
        icon: const Icon(Icons.add),
        label: const Text('Publicar'),
      ),
    );
  }
}

Future<void> confirmarEliminacion(
  BuildContext context,
  String tipo,
  Future<void> Function() eliminar,
) async {
  final confirmado = await showDialog<bool>(
    context: context,
    builder: (dialogContext) => AlertDialog(
      title: const Text('Confirmar eliminación'),
      content: Text(
        '¿Deseas eliminar este $tipo? Esta acción no se puede deshacer.',
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.pop(dialogContext, false),
          child: const Text('Cancelar'),
        ),
        FilledButton(
          onPressed: () => Navigator.pop(dialogContext, true),
          child: const Text('Eliminar'),
        ),
      ],
    ),
  );
  if (confirmado == true && context.mounted) {
    try {
      await eliminar();
    } catch (error) {
      if (context.mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('No se pudo eliminar: $error')),
        );
      }
    }
  }
}

// Campo reutilizable con validación de espacios en blanco.
class CampoTexto extends StatelessWidget {
  final TextEditingController controller;
  final String etiqueta;
  final int maxLineas;
  final TextInputType? teclado;

  const CampoTexto({
    super.key,
    required this.controller,
    required this.etiqueta,
    this.maxLineas = 1,
    this.teclado,
  });

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 14),
      child: TextFormField(
        controller: controller,
        maxLines: maxLineas,
        keyboardType: teclado,
        decoration: InputDecoration(
          labelText: etiqueta,
          border: const OutlineInputBorder(),
        ),
        validator: (valor) => valor == null || valor.trim().isEmpty
            ? 'El campo $etiqueta es obligatorio'
            : null,
      ),
    );
  }
}

class FormularioBoletin extends StatefulWidget {
  final PublinfoStore store;
  const FormularioBoletin({super.key, required this.store});

  @override
  State<FormularioBoletin> createState() => _FormularioBoletinState();
}

class _FormularioBoletinState extends State<FormularioBoletin> {
  final formKey = GlobalKey<FormState>();
  final titulo = TextEditingController();
  final resumen = TextEditingController();
  final zona = TextEditingController();
  final categoria = TextEditingController();

  @override
  void dispose() {
    titulo.dispose();
    resumen.dispose();
    zona.dispose();
    categoria.dispose();
    super.dispose();
  }

  Future<void> guardar() async {
    if (!formKey.currentState!.validate()) return;
    try {
      await widget.store.agregarBoletin(
        titulo: titulo.text,
        resumen: resumen.text,
        zona: zona.text,
        categoria: categoria.text,
      );
      if (mounted) Navigator.pop(context);
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('No se pudo guardar: $error')),
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Nuevo boletín')),
      body: Form(
        key: formKey,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            CampoTexto(controller: titulo, etiqueta: 'Título'),
            CampoTexto(controller: resumen, etiqueta: 'Resumen', maxLineas: 4),
            CampoTexto(controller: zona, etiqueta: 'Zona'),
            CampoTexto(controller: categoria, etiqueta: 'Categoría'),
            FilledButton(
              onPressed: guardar,
              child: const Text('Guardar boletín'),
            ),
          ],
        ),
      ),
    );
  }
}

}

class _FormularioAnuncioState extends State<FormularioAnuncio> {
  final formKey = GlobalKey<FormState>();
  final titulo = TextEditingController();
  final descripcion = TextEditingController();
  String? comercioId;

  @override
  void dispose() {
    titulo.dispose();
    descripcion.dispose();
    super.dispose();
  }

  Future<void> guardar() async {
    if (!formKey.currentState!.validate()) return;
    final id = comercioId;
    if (id == null) return;
    try {
      await widget.store.agregarAnuncio(
        comercioId: id,
        titulo: titulo.text,
        descripcion: descripcion.text,
      );
      if (mounted) Navigator.pop(context);
    } catch (error) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('No se pudo guardar: $error')),
        );
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Nuevo anuncio')),
      body: Form(
        key: formKey,
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [
            DropdownButtonFormField<String>(
              initialValue: comercioId,
              isExpanded: true,
              decoration: const InputDecoration(
                labelText: 'Comercio',
                border: OutlineInputBorder(),
              ),
              items: [
                for (final comercio in widget.store.comercios)
                  DropdownMenuItem(
                    value: comercio.id,
                    child: Text(comercio.nombre),
                  ),
              ],
              onChanged: (valor) => setState(() => comercioId = valor),
              validator: (valor) =>
                  valor == null ? 'Selecciona un comercio' : null,
            ),
            const SizedBox(height: 14),
            CampoTexto(controller: titulo, etiqueta: 'Título'),
            CampoTexto(
              controller: descripcion,
              etiqueta: 'Descripción',
              maxLineas: 4,
            ),
            FilledButton(
              onPressed: guardar,
              child: const Text('Guardar anuncio'),
            ),
          ],
        ),
      ),
    );
  }
}
